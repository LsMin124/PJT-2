        return "duplicate"                       # 오류 없이 무시. executor는 state를 다시 내지 않는다
    if incoming.order_update_id < cur.order_update_id:
        raise OrderError(...)                    # → orderUpdateError
    self._stitch(incoming); return "stitched"

inbox = queue.Queue()
# VDA: 구독 uagv/v2/+/+/{state,connection,factsheet} → inbox.put((topic, json))
# BIZ: 구독 biz/v1/order (QoS1) → inbox
# SPB: Edge Node fms-core — NBIRTH(Fms/RunId, Fms/Tick, Fms/RobotsOnline, Fms/OpenTasks, Fms/DoneTasks)
outbox = {}                                      # robot → (order, states_seen, tries)  ← 틱이 아니라 state 횟수
while running:
    snapshot = drain(inbox)                      # 로봇별 최신 state·접속·팩트시트, 새 주문. state 1건마다 states_seen += 1
    decisions = kernel.step(world, now, snapshot) # 순수 함수
    for robot, order in decisions.orders:        # orderUpdateId 스티칭 규칙 적용
        publish(f"uagv/v2/{robot.mfr}/{robot.serial}/order", order, qos=0)
        outbox[robot] = (order, 0, 0)
    for robot, (order, seen, tries) in outbox.items():          # 계약 4.3 재발행 — state 도착 횟수 기준
        if state_reflects(snapshot, robot, order): continue
        if seen >= 2:                                           # 첫 1건은 stale일 수 있어 2건까지 기다린다
            if tries >= 5: mark_robot_error(robot); continue    # state 10건 동안 미반영
            publish(order_topic(robot), order, qos=0); outbox[robot] = (order, 0, tries + 1)
        if reconnected(snapshot, robot): publish(order_topic(robot), order, qos=0)
        if online(snapshot, robot) and wall_since_last_state(robot) > 30: mark_broken(robot)   # 무소식 안전장치
    for ev in decisions.events:
        publish("biz/v1/event", json.dumps(ev), qos=1)          # 일반 JSON — WMS·Host가 읽는다
    node.publish_data({"Fms/Tick": tick, "Fms/RobotsOnline": online,
                       "Fms/OpenTasks": open_n, "Fms/DoneTasks": done_n})   # 값만, RBE
    sleep_until(next_tick)                       # 500 ms

리허설 — 한 바퀴와 장애 다섯

  1. 로컬 EMQX + 관측 창(tap, mqttx). Host 기동 → STATE online.
  2. GPU 서버 sim-runner(systemd) → NBIRTH, Runner/State=IDLE.
  3. Host Start(iw.hub 3대) → Isaac ready → ONLINE·factsheet ×3 → DBIRTH ×3 → RUNNING.
  4. fms-core 기동 → NBIRTH Fms/RunId 일치 확인 → 그 뒤에만 WMS 수요 생성. 먼저 biz/v1/order가 나오면 위반.
  5. order → state.lastNodeId 진행(Robot/LastNodeId), biz/v1/event 관찰. 2분 주행.
  6. Stop → OFFLINE ×3 → DDEATH ×3 → IDLE, Isaac 프로세스 0개.
장애기대
에이전트 kill -945 s 안에 그 시리얼만 CONNECTIONBROKEN. 재기동 → ONLINE·factsheet → outbox 재발행 → 같은 orderUpdateId면 무시(orderError가 나오면 05 수정 누락)
sim-runner kill -9NDEATH(bdSeq 일치) → 디바이스 stale → 하트비트 3회 결측 → 런 실패. systemd 재기동 → 잔여 정리 → bdSeq+1 NBIRTH
Host 재기동STATE offline → online → Edge들이 NBIRTH·DBIRTH 재발행
EMQX 재기동전부 재접속, retained 유지(볼륨), Edge는 bdSeq+1 NBIRTH
fms-core kill -9NDEATH → 런 실패 → Stop. 그 사이 주문 유실은 런 무효로 흡수
