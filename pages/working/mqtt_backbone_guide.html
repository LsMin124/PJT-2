문서와 규격의 차이

# BIZ: 구독 biz/v1/order (QoS1) → inbox

outbox = {}                                      # robot → (order, sent_tick, tries)
while running:

        if reconnected(snapshot, robot): publish(order_topic(robot), order, qos=0)
    for ev in decisions.events:
